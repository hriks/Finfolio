# Expense Manager — Plan 1: Foundation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold the React Native project, set up a tested SQLite data layer with migrations, and implement `ExpenseService` and `BudgetService` end-to-end in pure JS with full test coverage.

**Architecture:** Bare React Native CLI project. SQLite via `op-sqlite` on device. A `Database` interface abstracts the driver so unit tests can run against `better-sqlite3` in Node. All business logic in pure TypeScript modules; native code untouched in this plan.

**Tech Stack:** React Native 0.74+ (bare CLI), TypeScript, `@op-engineering/op-sqlite` (device), `better-sqlite3` (test), Jest, `ts-jest`, `date-fns`, `react-native-uuid`.

Create `src/data/better-sqlite3-driver.ts`:
```typescript
import BetterSqlite3 from 'better-sqlite3';
import type { Database, RunResult } from './database';

export const createBetterSqlite3Driver = (path: string = ':memory:'): Database => {
  const db = new BetterSqlite3(path);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  return {
    exec: (sql) => db.exec(sql),
    run: (sql, params = []) => {
      const info = db.prepare(sql).run(...params);
      return { changes: info.changes, lastInsertRowid: info.lastInsertRowid } as RunResult;
    },
    get: (sql, params = []) => db.prepare(sql).get(...params) as never,
    all: (sql, params = []) => db.prepare(sql).all(...params) as never,
    transaction: (fn) => db.transaction(fn)(),
    close: () => db.close(),
  };
};
```

```typescript
import type { Database } from './database';
import { migrations } from './migrations';

export const runMigrations = (db: Database): void => {
  const row = db.get<{ user_version: number }>('PRAGMA user_version');
  const current = row?.user_version ?? 0;
  db.transaction(() => {
    for (const m of migrations) {
      if (m.version > current) {
        db.exec(m.up);
        db.exec(`PRAGMA user_version = ${m.version}`);
      }
    }
  });
};
```

```typescript
import type { Expense, ExpenseSource, ExpenseStatus } from '../types/domain';

const COLUMNS =
  'id, amount_minor, currency, occurred_at, created_at, updated_at, merchant_raw, merchant_norm, category_id, source, source_ref, source_msg, confidence, status, note, photo_path, dedup_key, verified_by';

const rowToExpense = (r: Record<string, unknown>): Expense => ({
  id: r.id as string,
  amountMinor: r.amount_minor as number,
  currency: r.currency as string,
  occurredAt: r.occurred_at as number,
  createdAt: r.created_at as number,
  updatedAt: r.updated_at as number,
  merchantRaw: (r.merchant_raw as string) ?? null,
  merchantNorm: (r.merchant_norm as string) ?? null,
  categoryId: (r.category_id as string) ?? null,
  source: r.source as ExpenseSource,
  sourceRef: (r.source_ref as string) ?? null,
  sourceMsg: (r.source_msg as string) ?? null,
  confidence: r.confidence as number,
  status: r.status as ExpenseStatus,
  note: (r.note as string) ?? null,
  photoPath: (r.photo_path as string) ?? null,
  dedupKey: (r.dedup_key as string) ?? null,
  verifiedBy: r.verified_by as number,
});

export interface InsertExpenseInput {
  amountMinor: number;
  currency?: string;
  occurredAt: number;
  merchantRaw: string | null;
  merchantNorm: string | null;
  categoryId: string | null;
  source: ExpenseSource;
  sourceRef?: string | null;
  sourceMsg?: string | null;
  confidence?: number;
  status?: ExpenseStatus;
  note?: string | null;
  photoPath?: string | null;
  dedupKey?: string | null;
  verifiedBy?: number;
}

export interface ListOptions {
  includeVoid?: boolean;
  limit?: number;
  offset?: number;
}

export interface ExpenseService {
  insert(input: InsertExpenseInput): Expense;
  get(id: string): Expense | undefined;
  list(opts?: ListOptions): Expense[];
  setStatus(id: string, status: ExpenseStatus): Expense;
}

export const createExpenseService = (deps: { db: Database; clock: Clock }): ExpenseService => {
  const { db, clock } = deps;

  const insert: ExpenseService['insert'] = (input) => {
    const id = newId();
    const now = clock.now();
    db.run(
      `INSERT INTO expenses (${COLUMNS})
       VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
      [
        id,
        input.amountMinor,
        input.currency ?? 'INR',
        input.occurredAt,
        now,
        now,
        input.merchantRaw,
        input.merchantNorm,
        input.categoryId,
        input.source,
        input.sourceRef ?? null,
        input.sourceMsg ?? null,
        input.confidence ?? 1.0,
        input.status ?? 'active',
        input.note ?? null,
        input.photoPath ?? null,
        input.dedupKey ?? null,
        input.verifiedBy ?? 1,
      ],
    );
    return get(id)!;
  };

  const get = (id: string): Expense | undefined => {
    const row = db.get<Record<string, unknown>>(`SELECT ${COLUMNS} FROM expenses WHERE id = ?`, [id]);
    return row ? rowToExpense(row) : undefined;
  };

  const list: ExpenseService['list'] = (opts = {}) => {
    const where = opts.includeVoid ? '' : "WHERE status != 'void'";
    const limit = opts.limit ?? 100;
    const offset = opts.offset ?? 0;
    const rows = db.all<Record<string, unknown>>(
      `SELECT ${COLUMNS} FROM expenses ${where} ORDER BY occurred_at DESC LIMIT ? OFFSET ?`,
      [limit, offset],
    );
    return rows.map(rowToExpense);
  };

  const setStatus: ExpenseService['setStatus'] = (id, status) => {
    const now = clock.now();
    const res = db.run('UPDATE expenses SET status = ?, updated_at = ? WHERE id = ?', [status, now, id]);
    if (res.changes === 0) throw new Error(`expense not found: ${id}`);
    return get(id)!;
  };

  return { insert, get, list, setStatus };
};
```

```typescript
const rowToBudget = (r: Record<string, unknown>): Budget => ({
  id: r.id as string,
  period: r.period as BudgetPeriod,
  amountMinor: r.amount_minor as number,
  categoryId: (r.category_id as string) ?? null,

  id?: string;
  period: BudgetPeriod;
  amountMinor: number;
  categoryId: string | null;
  alertPct: number[];
}

export interface BudgetService {
  upsert(input: UpsertBudgetInput): Budget;
  get(id: string): Budget | undefined;
  list(): Budget[];
  delete(id: string): void;
}

export const createBudgetService = (deps: { db: Database; clock: Clock }): BudgetService => {
  const { db, clock } = deps;

  const get = (id: string): Budget | undefined => {

    return row ? rowToBudget(row) : undefined;
  };

    const pct = input.alertPct.join(',');
    if (input.id) {

        `UPDATE budgets SET period = ?, amount_minor = ?, category_id = ?, alert_pct = ?, updated_at = ? WHERE id = ?`,
        [input.period, input.amountMinor, input.categoryId, pct, now, input.id],

       VALUES (?, ?, ?, ?, ?, ?, ?)`,
      [id, input.period, input.amountMinor, input.categoryId, pct, now, now],
    );
    return get(id)!;
  };

  const list = (): Budget[] =>
    db.all<Record<string, unknown>>('SELECT * FROM budgets ORDER BY created_at ASC').map(rowToBudget);

  const del = (id: string): void => {
    db.run('DELETE FROM budget_alert_state WHERE budget_id = ?', [id]);

  };
```

```typescript
  const periodRange = (period: BudgetPeriod, atMs: number): { start: number; end: number } => {
    const at = new Date(atMs);
    switch (period) {
      case 'day':   return { start: startOfDay(at).getTime(),   end: endOfDay(at).getTime() };
      case 'week':  return { start: startOfWeek(at, { weekStartsOn: 1 }).getTime(), end: endOfWeek(at, { weekStartsOn: 1 }).getTime() };
      case 'month': return { start: startOfMonth(at).getTime(), end: endOfMonth(at).getTime() };
      case 'year':  return { start: startOfYear(at).getTime(),  end: endOfYear(at).getTime() };
    }
  };

  const rollup: BudgetService['rollup'] = (id) => {
    const b = get(id);
    if (!b) throw new Error(`budget not found: ${id}`);
    const { start, end } = periodRange(b.period, clock.now());
    const catClause = b.categoryId ? 'AND category_id = ?' : '';
    const params: unknown[] = [start, end];
    if (b.categoryId) params.push(b.categoryId);
    const row = db.get<{ s: number }>(
      `SELECT COALESCE(SUM(amount_minor), 0) AS s
         FROM expenses
        WHERE status = 'active'
          AND occurred_at BETWEEN ? AND ?
          ${catClause}`,
      params,
    );
    const spent = row?.s ?? 0;
    const remaining = Math.max(b.amountMinor - spent, 0);
    const pct = b.amountMinor === 0 ? 0 : Math.round((spent / b.amountMinor) * 100);
    return { spentMinor: spent, remainingMinor: remaining, pct, rangeStart: start, rangeEnd: end };
  };
```

```typescript
    const d = new Date(atMs);
    switch (period) {
      case 'day':   return format(d, 'yyyy-MM-dd');
      case 'week':  return format(startOfWeek(d, { weekStartsOn: 1 }), "yyyy-'W'II");
      case 'month': return format(d, 'yyyy-MM');

    const r = rollup(id);

    const stateRow = db.get<{ fired_pcts: string }>(

    const fired = new Set((stateRow?.fired_pcts ?? '').split(',').filter(Boolean).map(Number));
    const newlyCrossed: number[] = [];
    for (const pct of b.alertPct.slice().sort((a, c) => a - c)) {
```

```typescript
import { open, type DB } from '@op-engineering/op-sqlite';
import type { Database, RunResult } from './database';

export interface OpSqliteOptions {
  name: string;       // file name, e.g. 'expense-manager.db'
  location?: string;  // optional override
}

export const createOpSqliteDriver = (opts: OpSqliteOptions): Database => {
  const db: DB = open({ name: opts.name, location: opts.location });
  db.execute('PRAGMA journal_mode = WAL');
  db.execute('PRAGMA foreign_keys = ON');
  return {
    exec: (sql) => { db.execute(sql); },
    run: (sql, params = []) => {
      const res = db.execute(sql, params as never[]);
      return { changes: res.rowsAffected ?? 0, lastInsertRowid: res.insertId ?? 0 } as RunResult;
    },
    get: (sql, params = []) => {
      const res = db.execute(sql, params as never[]);
      return (res.rows?.[0] as never) ?? undefined;
    },
    all: (sql, params = []) => {
      const res = db.execute(sql, params as never[]);
      return (res.rows ?? []) as never;
    },
    transaction: (fn) => {
      db.execute('BEGIN');
      try {
        const out = fn();
        db.execute('COMMIT');
        return out;
      } catch (e) {
        db.execute('ROLLBACK');
        throw e;
      }
    },
    close: () => db.close(),
  };
};
```
